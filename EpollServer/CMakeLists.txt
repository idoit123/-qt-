cmake_minimum_required(VERSION 3.16)
project(epoll_chat_server LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "epoll_chat_server requires Linux")
endif()
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
option(CHAT_WITH_REDIS "Build optional Redis friend cache" ON)
option(CHAT_WITH_VIDEO "Enable OpenCV video motion detection" OFF)
add_library(chat_database src/database.cpp src/redis_cache.cpp)
target_include_directories(chat_database PUBLIC src)
target_link_libraries(chat_database PUBLIC Threads::Threads SQLite::SQLite3)
target_compile_options(chat_database PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CHAT_WITH_REDIS)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(HIREDIS REQUIRED IMPORTED_TARGET hiredis)
    target_link_libraries(chat_database PRIVATE PkgConfig::HIREDIS)
    target_compile_definitions(chat_database PRIVATE CHAT_WITH_REDIS)
endif()
add_executable(epoll_chat_server src/main.cpp src/epoll_server.cpp src/file_store.cpp)
target_link_libraries(epoll_chat_server PRIVATE chat_database)
target_compile_options(epoll_chat_server PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(epoll_chat_server PRIVATE _GNU_SOURCE)
if(CHAT_WITH_VIDEO)
    find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs video)
    target_sources(epoll_chat_server PRIVATE src/video_processor.cpp src/motion_regions.cpp)
    target_include_directories(epoll_chat_server PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(epoll_chat_server PRIVATE ${OpenCV_LIBS})
    target_compile_definitions(epoll_chat_server PRIVATE CHAT_WITH_VIDEO)
endif()
enable_testing()
if(CHAT_WITH_VIDEO)
    add_executable(motion_regions_test tests/motion_regions_test.cpp src/motion_regions.cpp)
    target_include_directories(motion_regions_test PRIVATE src ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(motion_regions_test PRIVATE ${OpenCV_LIBS})
    target_compile_options(motion_regions_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME motion_regions_test COMMAND motion_regions_test)
endif()
add_executable(epoll_core_test tests/core_test.cpp)
target_include_directories(epoll_core_test PRIVATE src)
target_compile_options(epoll_core_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME epoll_core_test COMMAND epoll_core_test)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    if(CHAT_WITH_VIDEO)
        add_test(NAME video_integration_test
            COMMAND ${CMAKE_COMMAND} -E env --unset=CHAT_REDIS_HOST
                ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/video_integration_test.py
                $<TARGET_FILE:epoll_chat_server>)
        set_tests_properties(video_integration_test PROPERTIES TIMEOUT 45)
    endif()
    add_test(NAME file_integration_test
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/file_integration_test.py
                $<TARGET_FILE:epoll_chat_server>)
    set_tests_properties(file_integration_test PROPERTIES TIMEOUT 120)
    add_test(NAME epoll_integration_test
        COMMAND ${CMAKE_COMMAND} -E env --unset=CHAT_REDIS_HOST
                ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration_test.py
                $<TARGET_FILE:epoll_chat_server>)
endif()
if(CHAT_WITH_REDIS)
    add_executable(redis_cache_probe tests/redis_cache_probe.cpp)
    target_link_libraries(redis_cache_probe PRIVATE chat_database)
    target_compile_options(redis_cache_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
    find_program(REDIS_SERVER redis-server)
    find_program(REDIS_CLI redis-cli)
    if(Python3_Interpreter_FOUND AND REDIS_SERVER AND REDIS_CLI)
        add_test(NAME redis_cache_integration_test
            COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/redis_cache_test.py
                $<TARGET_FILE:redis_cache_probe> $<TARGET_FILE:epoll_chat_server>
                ${REDIS_SERVER} ${REDIS_CLI})
        set_tests_properties(redis_cache_integration_test PROPERTIES TIMEOUT 60)
    else()
        message(WARNING "Redis integration test requires Python3, redis-server and redis-cli")
    endif()
endif()
